\begin{align}
m_{\mathrm{surffront}}(x,y)
&=
\frac{Q f_f}{\pi \alpha_f \beta_f}
\exp\left[
-3\left(
\frac{x^2}{\alpha_f^2}
+
\frac{y^2}{\beta_f^2}
\right)
\right]
\tag{4}
\\
m_{\mathrm{surfback}}(x,y)
&=
\frac{Q f_r}{\pi \alpha_r \beta_r}
\exp\left[
-3\left(
\frac{x^2}{\alpha_r^2}
+
\frac{y^2}{\beta_r^2}
\right)
\right]
\tag{5}
\\
m_{\mathrm{surf}}(x,y)
&=
m_{\mathrm{surffront}}(x,y)
+
m_{\mathrm{surfback}}(x,y)
\tag{6}
\end{align}
